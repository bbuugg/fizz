# Dissolve Upload Tile — React

An attachment assembles from grains as it uploads, then disintegrates on failure, cancellation or removal.

Source: https://dissolve.rselmi.com/
Preserve embedded font and third-party license notices.

## Install
```sh
npm install react react-dom
```

Keep all four files in one folder. The framework's normal build tool compiles TypeScript, Vue or Svelte.

## UploadTray.tsx
```tsx
"use client";
import { useEffect, useRef, type CSSProperties } from "react";
import { createUploadTile, createUploadTray, mountUploadTray, type UploadTileProps, type UploadTrayProps, type UploadTrayOptions } from "./UploadTrayEngine.js";

export type { UploadFile, UploadFn, RemoveReason, UploadTileProps, UploadTrayProps, UploadTrayOptions } from "./UploadTrayEngine.js";

type Appearance = { className?: string; style?: CSSProperties };

export function UploadTray({ className, style, ...props }: Appearance & (UploadTrayProps | { options?: UploadTrayOptions })) {
  const host = useRef<HTMLDivElement>(null);
  const controller = useRef<{ update(options: UploadTrayOptions): void; destroy(): void } | null>(null);
  const direct = "files" in props;
  const options = direct ? props : props.options;
  useEffect(() => {
    const instance = direct ? createUploadTray(host.current!, props) : mountUploadTray(host.current!, options);
    controller.current = instance;
    return () => { instance.destroy(); controller.current = null; };
    // Props update below while every existing file keeps its own upload.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [direct]);
  useEffect(() => controller.current?.update(options ?? {}), [options]);
  return <div ref={host} className={className} style={style} data-motion-component="upload-tile" />;
}

export function UploadTile({ className, style, ...props }: Appearance & UploadTileProps) {
  const host = useRef<HTMLDivElement>(null);
  const controller = useRef<ReturnType<typeof createUploadTile> | null>(null);
  const { id, name, size, uri } = props.file;
  useEffect(() => {
    const instance = createUploadTile(host.current!, props);
    controller.current = instance;
    return () => { instance.destroy(); controller.current = null; };
    // A different file owns a different upload. Callbacks update below.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [id, name, size, uri]);
  useEffect(() => controller.current?.update(props), [props]);
  return <div ref={host} className={className} style={style} />;
}

function bezier(x1, y1, x2, y2) {
  const cx = 3 * x1, bx = 3 * (x2 - x1) - cx, ax = 1 - cx - bx;
  const cy = 3 * y1, by = 3 * (y2 - y1) - cy, ay = 1 - cy - by;
  const X = (t) => ((ax * t + bx) * t + cx) * t;
  const dX = (t) => (3 * ax * t + 2 * bx) * t + cx;
  const Y = (t) => ((ay * t + by) * t + cy) * t;
  return (x) => {
    if (x <= 0) return 0;
    if (x >= 1) return 1;
    let t = x;
    for (let i = 0; i < 8; i++) {
      const e = X(t) - x, d = dX(t);
      if (Math.abs(e) < 1e-6 || Math.abs(d) < 1e-6) break;
      t -= e / d;
    }
    if (t < 0 || t > 1 || Math.abs(X(t) - x) > 1e-4) {
      let lo = 0, hi = 1;
      t = x;
      for (let i = 0; i < 40; i++) {
        if (X(t) < x) lo = t;
        else hi = t;
        t = (lo + hi) / 2;
      }
    }
    return Y(t);
  };
}

function scope() {
  const timers = /* @__PURE__ */ new Set(), handles = /* @__PURE__ */ new Set(), fns = [];
  const sc = {
    alive: true,
    wait: (ms) => new Promise((res) => {
      const t = setTimeout(() => {
        timers.delete(t);
        if (sc.alive) res();
      }, ms);
      timers.add(t);
    }),
    track(h) {
      handles.add(h);
      h.done.then(() => handles.delete(h));
      return h;
    },
    onDispose(fn) {
      fns.push(fn);
    },
    dispose() {
      if (!sc.alive) return;
      sc.alive = false;
      timers.forEach(clearTimeout);
      handles.forEach((h) => h.cancel());
      fns.forEach((f) => f());
    }
  };
  return sc;
}
function value(sc, v0, onChange) {
  let id = 0;
  const a = {
    v: v0,
    vel: 0,
    set(x) {
      id++;
      a.v = x;
      a.vel = 0;
      onChange();
    },
    to(to, ms, ease = CALM, delay = 0) {
      const my = ++id, from = a.v;
      return new Promise((res) => {
        const t0 = performance.now() + delay;
        let lastT = t0;
        const f = (now) => {
          if (my !== id) return res(false);
          if (!sc.alive) return;
          if (now < t0) return void requestAnimationFrame(f);
          const k = ms > 0 ? Math.min(1, (now - t0) / ms) : 1;
          const prev = a.v;
          a.v = from + (to - from) * ease(k);
          const dt = (now - lastT) / 1e3;
          a.vel = dt > 0 ? (a.v - prev) / dt : 0;
          lastT = now;
          onChange();
          if (k < 1) requestAnimationFrame(f);
          else {
            a.vel = 0;
            res(true);
          }
        };
        requestAnimationFrame(f);
      });
    },
    spring(to, { stiffness = 260, damping = 31, mass = 1, eps = 0.02 } = {}) {
      const my = ++id;
      return new Promise((res) => {
        let last = performance.now();
        const f = (now) => {
          if (my !== id) return res(false);
          if (!sc.alive) return;
          let dt = Math.min(0.064, (now - last) / 1e3);
          last = now;
          while (dt > 0) {
            const h = Math.min(1 / 480, dt);
            a.vel += (-stiffness * (a.v - to) - damping * a.vel) / mass * h;
            a.v += a.vel * h;
            dt -= h;
          }
          if (Math.abs(a.v - to) < eps && Math.abs(a.vel) < eps * 10) {
            a.v = to;
            a.vel = 0;
            onChange();
            return res(true);
          }
          onChange();
          requestAnimationFrame(f);
        };
        requestAnimationFrame(f);
      });
    }
  };
  return a;
}

function cloudScale(P) {
  const s = Math.sqrt(P.w * P.h) / 160;
  return s < 0.6 ? 0.6 : s > 1.8 ? 1.8 : s;
}
function burstOut(d, P, strength, extraIdx, randomDir) {
  return (k) => {
    const dx = d[k] - P.ox, dy = d[k + 1] - P.oy;
    const dist = Math.sqrt(dx * dx + dy * dy);
    let ux, uy;
    if (dist < 1) {
      const an = (randomDir ? Math.random() : d[k + 5]) * Math.PI * 2;
      ux = Math.cos(an);
      uy = Math.sin(an);
    } else {
      ux = dx / dist;
      uy = dy / dist;
    }
    const push = (30 + d[k + 3] * 110) * strength;
    const swirl = (d[k + 4] - 0.5) * 30;
    d[k + extraIdx] = ux * push - uy * swirl;
    d[k + extraIdx + 1] = uy * push + ux * swirl;
  };
}
function flingAxis(P) {
  let vx = P.vx, vy = P.vy, sp = Math.hypot(vx, vy);
  if (sp < 1) {
    vx = 1e3;
    vy = 0;
    sp = 1e3;
  }
  const ux = vx / sp, uy = vy / sp;
  return { ux, uy, nx: -uy, ny: ux, sp, k: Math.min(2, Math.max(0.4, sp / 1500)) };
}
function maxTargetDist(P) {
  let m = 0;
  for (let i = 0; i < 4; i++) {
    const dd = Math.hypot(P.x + (i & 1 ? P.w : 0) - P.tx, P.y + (i & 2 ? P.h : 0) - P.ty);
    if (dd > m) m = dd;
  }
  return m || 1;
}

function clipRect(poly, w, h) {
  const planes = [(p) => p.x, (p) => w - p.x, (p) => p.y, (p) => h - p.y];
  let out = poly;
  for (const f of planes) {
    if (!out.length) break;
    const inp = out;
    out = [];
    for (let i = 0; i < inp.length; i++) {
      const a = inp[i], b = inp[(i + 1) % inp.length], fa = f(a), fb = f(b);
      if (fa >= 0) out.push(a);
      if (fa >= 0 !== fb >= 0) {
        const u = fa / (fa - fb);
        out.push({ x: a.x + (b.x - a.x) * u, y: a.y + (b.y - a.y) * u });
      }
    }
  }
  return out;
}
function centroid(poly) {
  let A = 0, cx = 0, cy = 0;
  for (let i = 0; i < poly.length; i++) {
    const a = poly[i], b = poly[(i + 1) % poly.length], c = a.x * b.y - b.x * a.y;
    A += c;
    cx += (a.x + b.x) * c;
    cy += (a.y + b.y) * c;
  }
  A /= 2;
  if (Math.abs(A) < 1e-9) return { A: 0, x: poly[0].x, y: poly[0].y };
  return { A, x: cx / (6 * A), y: cy / (6 * A) };
}
function polarShards(w, h, ox, oy, maxD, q, S) {
  const base = Math.random() * Math.PI * 2, sector = Math.PI * 2 / S;
  const rMin = Math.min(14, Math.max(8, maxD * 0.04)), rEnd = maxD / Math.cos(sector / 2) * 1.05;
  const radii = [0];
  for (let r = rMin; ; r *= q) {
    radii.push(r);
    if (r >= rEnd) break;
  }
  const rings = radii.length, grid = [];
  for (let k = 0; k < rings; k++) {
    const row = [], last = k === rings - 1;
    for (let j = 0; j < S; j++) {
      if (k === 0) {
        row.push({ x: ox, y: oy });
        continue;
      }
      const th = base + j * sector + (last ? 0 : (Math.random() - 0.5) * 0.3 * sector);
      const r = last ? radii[k] : radii[k] * (1 + (Math.random() - 0.5) * 0.12);
      row.push({ x: ox + Math.cos(th) * r, y: oy + Math.sin(th) * r });
    }
    grid.push(row);
  }
  const out = [];
  const keep = (tri) => {
    const c = clipRect(tri, w, h);
    if (c.length >= 3 && Math.abs(centroid(c).A) > 1e-4) out.push(c);
  };
  const cross = (p, q2, r) => (q2.x - p.x) * (r.y - p.y) - (q2.y - p.y) * (r.x - p.x);
  for (let k = 0; k < rings - 1; k++)
    for (let j = 0; j < S; j++) {
      const j2 = (j + 1) % S, A = grid[k][j], B = grid[k][j2], C = grid[k + 1][j2], D = grid[k + 1][j];
      const acOk = cross(A, B, C) * cross(A, C, D) > 0, bdOk = cross(A, B, D) * cross(B, C, D) > 0;
      if (k === 0) keep([A, C, D]);
      else if (acOk && (!bdOk || Math.random() < 0.5)) {
        keep([A, B, C]);
        keep([A, C, D]);
      } else {
        keep([A, B, D]);
        keep([B, C, D]);
      }
    }
  return out;
}
function shatterMesh(w, h, ox, oy, target, fvx, fvy) {
  const cx = Math.min(w, Math.max(0, ox)), cy = Math.min(h, Math.max(0, oy));
  const maxD = Math.max(Math.hypot(cx, cy), Math.hypot(w - cx, cy), Math.hypot(cx, h - cy), Math.hypot(w - cx, h - cy), 1);
  let S = 10 + Math.floor(Math.random() * 5), q = 1.8;
  let polys = polarShards(w, h, cx, cy, maxD, q, S);
  for (let guard = 0; polys.length < target && guard < 80; guard++) {
    if (q > 1.42) q -= 0.06;
    else if (S < 64) S += 2;
    else break;
    polys = polarShards(w, h, cx, cy, maxD, q, S);
  }
  const shards = polys.map((poly) => {
    const g = centroid(poly);
    let ux = g.x - cx, uy = g.y - cy;
    const len = Math.hypot(ux, uy);
    if (len < 1e-3) {
      const a = Math.random() * Math.PI * 2;
      ux = Math.cos(a);
      uy = Math.sin(a);
    } else {
      ux /= len;
      uy /= len;
    }
    const dn = Math.min(1, len / maxD), speed = (120 + Math.random() * 200) * (1 - 0.6 * dn);
    const pts = new Float32Array(poly.length * 2);
    poly.forEach((p, i) => {
      const dx = p.x - g.x, dy = p.y - g.y, l = Math.hypot(dx, dy) || 1;
      pts[i * 2] = dx + dx / l * 0.5;
      pts[i * 2 + 1] = dy + dy / l * 0.5;
    });
    return {
      gx: g.x,
      gy: g.y,
      vx: ux * speed + fvx * 0.3,
      vy: uy * speed + fvy * 0.3,
      ux,
      uy,
      spin: Math.random() - 0.5,
      flip: Math.random(),
      delay: dn * 0.15 + Math.random() * 0.05,
      life: 0.8,
      pts
    };
  });
  return { shards, ox: cx, oy: cy, flashR: Math.min(170, Math.max(60, maxD * 0.55)) };
}

var simulateUpload = (file, onProgress, signal) => {
  const { outcome, duration } = file;
  return new Promise((resolve, reject) => {
    const start = performance.now();
    const stop = () => {
      clearInterval(tick);
      signal.removeEventListener("abort", abort);
    };
    const abort = () => {
      stop();
      reject(signal.reason);
    };
    const tick = setInterval(() => {
      const progress = (performance.now() - start) / duration;
      if (outcome === "failure" && progress >= 0.52) {
        stop();
        reject(new Error("Simulated upload failure"));
      } else if (progress >= 1) {
        stop();
        onProgress(1);
        resolve();
      } else {
        onProgress(progress);
      }
    }, 60);
    signal.addEventListener("abort", abort, { once: true });
    if (signal.aborted) abort();
  });
};

      done: ["Upload finished", "\u4E0A\u4F20\u5B8C\u6210"],
      cancelled: ["Upload cancelled", "\u5DF2\u53D6\u6D88\u4E0A\u4F20"],
      rejected: file.size > (options.maxSize ?? 25 * MB) ? ["Over size limit", "\u8D85\u51FA\u5927\u5C0F\u9650\u5236"] : ["Upload failed", "\u4E0A\u4F20\u5931\u8D25"],
      removed: ["Attachment removed", "\u5DF2\u79FB\u9664\u9644\u4EF6"]
    };
    status.textContent = `${descriptions[reason][zh() ? 1 : 0]} \xB7 ${file.name}`;
  }
  function add() {
    const id = ++nextId;
    samples = [...samples, {
      id: `sample-${id}`,
      name: outcome === "oversize" ? `archive-${id}.zip` : `notes-${id}.pdf`,
      size: outcome === "oversize" ? 32 * MB : 1.2 * MB,
      outcome,
      duration: 3e3
    }];
    tray.update({ files: samples });
  }
  function trayProps() {
    return {
      files: options.files ?? samples,
      upload: options.upload ?? (demo() ? simulateUpload : missingUploader),
      maxSize: options.maxSize ?? (demo() ? 25 * MB : void 0),
      onAdd: options.onAdd ?? (demo() && !options.preview ? add : void 0),
      onDone: (file) => {
        if (demo()) announce(file, "done");
        options.onDone?.(file);
      },
      onRemove: (file, reason) => {
        if (demo()) {
          announce(file, reason);
          samples = samples.filter((sample) => sample.id !== file.id);
          tray.update({ files: samples });
        }
        options.onRemove?.(file, reason);
      },
      gap: options.gap,
      size: options.size ?? (demo() ? options.preview ? 88 : 76 : void 0),
      haptics: options.haptics ?? false,
      tileClassName: options.tileClassName,
      locale: options.locale,
      reducedMotion: options.reducedMotion
    };
  }
  function syncDemo() {
    const labelIndex = zh() ? 1 : 0;
    controls.hidden = !demo() || !!options.preview;
    stage2.classList.toggle("library-demo", demo());
    stage2.dataset.preview = String(!!options.preview);
    stage2.lang = options.locale ?? "en";
    legend.textContent = zh() ? "\u6A21\u62DF\u7ED3\u679C" : "Simulated result";
    for (const radio of radios) {
      const value2 = radio.value;
      radio.checked = value2 === outcome;
      radio.nextElementSibling.textContent = outcomes[value2][labelIndex];
    }
  }
  syncDemo();
  const tray = createUploadTray(scope2.root.querySelector("[data-upload-tray]"), trayProps());
  for (const radio of radios) {
    scope2.on(radio, "change", () => {
      outcome = radio.value;
    });
  }
  scope2.addCleanup(() => {
    destroyed = true;
    tray.destroy();
  });
  return {
    ...scope2.controller,
    update(next) {
      if (destroyed) return;
      options = { ...options, ...next };
      scope2.update(surfaceOptions());
      syncDemo();
      tray.update(trayProps());
    }
  };
}

    addCleanup(cleanup: () => void) { cleanups.push(cleanup); },
    ready() { ready.splice(0).forEach(listener => { const event = new Event("DOMContentLoaded"); if (typeof listener === "function") listener.call(document, event); else listener.handleEvent(event); }); },
    point(event: { clientX: number; clientY: number }) { const rect = frame.getBoundingClientRect(); return { x: (event.clientX - rect.x) / scale, y: (event.clientY - rect.y) / scale }; },
    rect(node: Element) { const rect = node.getBoundingClientRect(); const origin = frame.getBoundingClientRect(); return { x: (rect.x - origin.x) / scale, y: (rect.y - origin.y) / scale, left: (rect.left - origin.left) / scale, top: (rect.top - origin.top) / scale, right: (rect.right - origin.left) / scale, bottom: (rect.bottom - origin.top) / scale, width: rect.width / scale, height: rect.height / scale }; },
  };
}
```
